cmake_minimum_required(VERSION 3.25)
project(uncertainty_feedback_archive_consumer VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Only the predeclared Release recipe is admitted")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" dependencies)
foreach(key IN ITEMS nlohmann_json cxx_compiler formatter zlib_library crypto_library)
    string(JSON path GET "${dependencies}" ${key} path)
    string(JSON expected GET "${dependencies}" ${key} sha256)
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned ${key} changed")
    endif()
    set(${key}_path "${path}")
endforeach()
file(REAL_PATH "${CMAKE_CXX_COMPILER}" compiler)
if(NOT compiler STREQUAL cxx_compiler_path)
    message(FATAL_ERROR "Exact declared C++ compiler required")
endif()
set(dependency_headers "${nlohmann_json_path}")
foreach(index RANGE 0 1)
    string(JSON path GET "${dependencies}" zlib_headers ${index} path)
    string(JSON expected GET "${dependencies}" zlib_headers ${index} sha256)
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned zlib header changed")
    endif()
    list(APPEND dependency_headers "${path}")
endforeach()
get_filename_component(json_include "${nlohmann_json_path}" DIRECTORY)
set(owned_cpp include/archive_consumer/core.h src/core.cpp src/archive.cpp src/request.cpp
    src/emission.cpp src/edits.cpp src/main.cpp src/controls.cpp)
file(GLOB_RECURSE actual_cpp RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS
    include/*.h src/*.cpp)
set(expected_cpp ${owned_cpp})
list(SORT actual_cpp)
list(SORT expected_cpp)
if(NOT actual_cpp STREQUAL expected_cpp)
    message(FATAL_ERROR "Owned archive consumer C++ census incomplete")
endif()
set(expected_closure ${owned_cpp} CMakeLists.txt .clang-format dependencies.json source-files.json
    provenance.json README.md qualification-plan.md)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" source_json)
string(JSON source_count LENGTH "${source_json}")
math(EXPR source_last "${source_count}-1")
set(source_closure ${dependency_headers})
set(supplied_closure)
foreach(index RANGE 0 ${source_last})
    string(JSON relative GET "${source_json}" ${index})
    list(APPEND supplied_closure "${relative}")
    get_filename_component(absolute "${CMAKE_CURRENT_SOURCE_DIR}/${relative}" ABSOLUTE)
    list(APPEND source_closure "${absolute}")
endforeach()
list(SORT supplied_closure)
list(SORT expected_closure)
if(NOT supplied_closure STREQUAL expected_closure)
    message(FATAL_ERROR "Exact owned source and protocol closure differs")
endif()
set(witness "#pragma once\n#include <cstdint>\nnamespace archive_consumer {\nstruct Source {const char* path; std::uintmax_t size; const char* sha256;};\ninline constexpr Source compiled_sources[] = {\n")
foreach(path IN LISTS source_closure)
    file(SHA256 "${path}" hash)
    file(SIZE "${path}" size)
    string(APPEND witness "{\"${path}\",${size},\"${hash}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
endforeach()
file(SHA256 "${compiler}" compiler_hash)
file(SIZE "${compiler}" compiler_size)
get_filename_component(legacy_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
string(APPEND witness "};\ninline constexpr const char* source_root_path=\"${legacy_root}\";\ninline constexpr const char* own_unit_path=\"${CMAKE_CURRENT_SOURCE_DIR}\";\ninline constexpr const char* compiler_path=\"${compiler}\";\ninline constexpr const char* compiler_sha256=\"${compiler_hash}\";\ninline constexpr std::uintmax_t compiler_size=${compiler_size};\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${witness}")
add_executable(uncertainty-feedback-archive-consumer src/core.cpp src/archive.cpp src/request.cpp
    src/emission.cpp src/edits.cpp src/main.cpp src/controls.cpp)
target_include_directories(uncertainty-feedback-archive-consumer PRIVATE include
    "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(uncertainty-feedback-archive-consumer SYSTEM PRIVATE "${json_include}")
target_link_libraries(uncertainty-feedback-archive-consumer PRIVATE
    "${crypto_library_path}" "${zlib_library_path}")
target_compile_options(uncertainty-feedback-archive-consumer PRIVATE -Wall -Wextra -Wpedantic
    -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(format_sources)
foreach(path IN LISTS owned_cpp)
    list(APPEND format_sources "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
add_custom_target(format-check COMMAND "${formatter_path}" --dry-run --Werror ${format_sources}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_test(NAME archive_consumer_synthetic_controls COMMAND uncertainty-feedback-archive-consumer
        controls --scratch "${CMAKE_CURRENT_BINARY_DIR}/synthetic-controls")
    set_tests_properties(archive_consumer_synthetic_controls PROPERTIES TIMEOUT 60)
endif()
