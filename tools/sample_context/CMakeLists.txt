cmake_minimum_required(VERSION 3.25)
project(sample_context VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Sample context qualification requires Release")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_program(SAMPLE_FORMATTER NAMES clang-format REQUIRED)
set(SAMPLE_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" census)
string(JSON count LENGTH "${census}")
math(EXPR last "${count}-1")
set(witness "#pragma once\n#include <cstdint>\nnamespace sample_compiled {\nstruct Source { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr Source sources[] = {\n")
set(owned_cpp)
set(all_files)
foreach(index RANGE ${last})
    string(JSON relative GET "${census}" ${index})
    get_filename_component(path "${relative}" REALPATH BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Sample source census file absent: ${path}")
    endif()
    list(APPEND all_files "${path}")
    file(SHA256 "${path}" hash)
    file(SIZE "${path}" size)
    string(APPEND witness "{\"${path}\",\"${hash}\",${size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    if(path MATCHES "^${CMAKE_CURRENT_SOURCE_DIR}/.*\\.(cpp|h)$")
        list(APPEND owned_cpp "${path}")
    endif()
endforeach()
list(LENGTH all_files before)
list(REMOVE_DUPLICATES all_files)
list(LENGTH all_files after)
if(NOT before EQUAL after)
    message(FATAL_ERROR "Sample source census duplicates")
endif()
file(GLOB_RECURSE actual_cpp CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT actual_cpp)
list(SORT owned_cpp)
if(NOT actual_cpp STREQUAL owned_cpp)
    message(FATAL_ERROR "Sample owned C++ census incomplete")
endif()
get_filename_component(compiler "${CMAKE_CXX_COMPILER}" REALPATH)
get_filename_component(formatter "${SAMPLE_FORMATTER}" REALPATH)
get_filename_component(root "../.." REALPATH BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
file(SHA256 "${compiler}" compiler_hash)
file(SHA256 "${formatter}" formatter_hash)
string(JOIN " " flags ${SAMPLE_FLAGS})
string(APPEND witness "};\ninline constexpr const char* root = \"${root}\";\ninline constexpr const char* compiler_path = \"${compiler}\";\ninline constexpr const char* compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* formatter_path = \"${formatter}\";\ninline constexpr const char* formatter_sha256 = \"${formatter_hash}\";\ninline constexpr const char* compile_flags = \"${flags}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sample_sources.h" "${witness}")
add_library(sample_context_core STATIC src/context.cpp src/runtime.cpp tests/controls.cpp)
target_include_directories(sample_context_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sample_context_core PUBLIC OpenSSL::Crypto)
target_compile_options(sample_context_core PUBLIC ${SAMPLE_FLAGS})
add_executable(sample-context src/main.cpp)
target_include_directories(sample-context PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sample-context PRIVATE sample_context_core)
add_custom_target(format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${formatter}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${formatter}" --dry-run --Werror ${owned_cpp}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_executable(sample-context-tests tests/main.cpp)
    target_link_libraries(sample-context-tests PRIVATE sample_context_core)
    add_test(NAME sample_context_contract_controls COMMAND sample-context-tests)
    set_tests_properties(sample_context_contract_controls PROPERTIES TIMEOUT 60)
endif()
