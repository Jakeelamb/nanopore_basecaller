cmake_minimum_required(VERSION 3.25)
project(prefix_feedback VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Prefix feedback requires explicit Release")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_program(FEEDBACK_FORMATTER NAMES clang-format REQUIRED)
set(FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/native-source-files.json" CENSUS)
string(JSON count LENGTH "${CENSUS}")
math(EXPR last "${count} - 1")
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace prefix_feedback {\nstruct CompiledSource { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr CompiledSource compiled_sources[] = {\n")
set(OWNED_CPP)
foreach(index RANGE ${last})
    string(JSON relative GET "${CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
    string(APPEND WITNESS "{\"${absolute}\",\"${hash}\",${size}},\n")
    if(absolute MATCHES "^${CMAKE_CURRENT_SOURCE_DIR}/.*\\.(cpp|h)$")
        list(APPEND OWNED_CPP "${absolute}")
    endif()
endforeach()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT OWNED_CPP)
list(SORT ACTUAL_CPP)
if(NOT OWNED_CPP STREQUAL ACTUAL_CPP)
    message(FATAL_ERROR "Explicit owned C++ census differs")
endif()
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_hash)
file(SHA256 "${FEEDBACK_FORMATTER}" formatter_hash)
string(APPEND WITNESS "};\ninline constexpr const char* compiler_path = \"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* formatter_path = \"${FEEDBACK_FORMATTER}\";\ninline constexpr const char* formatter_sha256 = \"${formatter_hash}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${WITNESS}")
set(SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../context_decoder")
add_library(prefix_feedback STATIC src/feedback.cpp "${SHARED}/src/decoder.cpp" "${SHARED}/src/guarded.cpp" "${SHARED}/src/io.cpp")
target_include_directories(prefix_feedback PUBLIC include "${SHARED}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(prefix_feedback PUBLIC ${FLAGS})
target_link_libraries(prefix_feedback PUBLIC OpenSSL::Crypto)
add_executable(prefix-feedback src/main.cpp)
target_link_libraries(prefix-feedback PRIVATE prefix_feedback)
add_custom_target(format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${FEEDBACK_FORMATTER}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${FEEDBACK_FORMATTER}" --dry-run --Werror ${OWNED_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_executable(prefix-feedback-tests tests/main.cpp tests/controls.cpp "${SHARED}/tests/guarded_controls.cpp")
    target_link_libraries(prefix-feedback-tests PRIVATE prefix_feedback)
    add_test(NAME causal_prefix_controls COMMAND prefix-feedback-tests)
    set_tests_properties(causal_prefix_controls PROPERTIES TIMEOUT 60)
endif()
