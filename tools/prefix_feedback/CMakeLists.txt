cmake_minimum_required(VERSION 3.25)
project(prefix_feedback VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FEEDBACK_BASE_ROOT "/home/jake/Projects/nanopore_basecaller-context-feedback-targeting-results" CACHE PATH "Read-only pinned predecessor")
set(FEEDBACK_CLANG_FORMAT "/usr/bin/clang-format" CACHE FILEPATH "Pinned formatter")
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
file(READ "${FEEDBACK_BASE_ROOT}/tools/uncertainty_feedback/dependencies.json" dependencies)
string(JSON json_header GET "${dependencies}" nlohmann_json path)
file(SHA256 "${FEEDBACK_BASE_ROOT}/tools/uncertainty_feedback/dependencies.json" dependency_hash)
if(NOT dependency_hash STREQUAL "f9a7bb78dfa8d5da15dc40541a2097f31ab9619d77602a62cbb3a2921a49ffd9")
    message(FATAL_ERROR "Read-only PR15 dependency manifest changed")
endif()
string(JSON json_expected GET "${dependencies}" nlohmann_json sha256)
file(SHA256 "${json_header}" json_actual)
if(NOT json_actual STREQUAL json_expected)
    message(FATAL_ERROR "Pinned JSON header changed")
endif()
get_filename_component(json_include "${json_header}" DIRECTORY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" source_manifest)
string(JSON source_count LENGTH "${source_manifest}")
math(EXPR source_last "${source_count} - 1")
set(closure)
set(owned)
foreach(index RANGE 0 ${source_last})
    string(JSON relative GET "${source_manifest}" ${index})
    list(APPEND closure "${relative}")
    if(relative MATCHES "\\.(cpp|h)$")
        list(APPEND owned "${relative}")
    endif()
endforeach()
file(GLOB_RECURSE actual RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS include/*.h src/*.cpp tests/*.cpp)
set(expected ${owned})
list(SORT actual)
list(SORT expected)
if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Owned C++ source census differs")
endif()
get_filename_component(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
list(APPEND closure source-files.json)
set(witness "#pragma once\nnamespace prefix_feedback {\nstruct CompiledSource { const char* path; const char* sha256; };\ninline constexpr CompiledSource compiled_sources[] = {\n")
foreach(relative IN LISTS closure)
    file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/${relative}" absolute)
    file(SHA256 "${absolute}" hash)
    string(APPEND witness "{\"${absolute}\",\"${hash}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
endforeach()
string(APPEND witness "{\"${json_header}\",\"${json_expected}\"},\n{\"${FEEDBACK_BASE_ROOT}/tools/uncertainty_feedback/dependencies.json\",\"${dependency_hash}\"},\n};\ninline constexpr const char* compiled_root=\"${source_root}\";\n}\n")
file(SHA256 "${FEEDBACK_CLANG_FORMAT}" formatter_hash)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${witness}")
add_executable(prefix-feedback src/io.cpp src/session.cpp src/location.cpp src/main.cpp src/bridge.cpp src/resource_evidence.cpp tests/controls.cpp tests/bridge_controls.cpp tests/resource_controls.cpp)
target_include_directories(prefix-feedback PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(prefix-feedback SYSTEM PRIVATE "${json_include}")
target_link_libraries(prefix-feedback PRIVATE OpenSSL::Crypto)
target_compile_options(prefix-feedback PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(format_sources)
foreach(relative IN LISTS owned)
    list(APPEND format_sources "${CMAKE_CURRENT_SOURCE_DIR}/${relative}")
endforeach()
add_custom_target(format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${FEEDBACK_CLANG_FORMAT}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${FEEDBACK_CLANG_FORMAT}" --dry-run --Werror ${format_sources} VERBATIM COMMAND_EXPAND_LISTS)
if(BUILD_TESTING)
    add_test(NAME prefix_feedback_controls COMMAND prefix-feedback controls)
    add_test(NAME prefix_bridge_controls COMMAND prefix-feedback bridge-controls)
    set_tests_properties(prefix_feedback_controls PROPERTIES TIMEOUT 120 WORKING_DIRECTORY "${source_root}")
    set_tests_properties(prefix_bridge_controls PROPERTIES TIMEOUT 120 WORKING_DIRECTORY "${source_root}")
endif()
