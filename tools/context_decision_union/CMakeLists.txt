cmake_minimum_required(VERSION 3.25)
project(context_decision_union VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(UNION_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(UNION_TESTING "${BUILD_TESTING}")
set(BUILD_TESTING OFF)
add_subdirectory("../context_decoder/study" "${CMAKE_CURRENT_BINARY_DIR}/study" EXCLUDE_FROM_ALL)
set(BUILD_TESTING "${UNION_TESTING}")
find_program(UNION_CLANG_FORMAT NAMES clang-format REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" UNION_CENSUS)
string(JSON count LENGTH "${UNION_CENSUS}")
math(EXPR last "${count} - 1")
set(UNION_FILES)
set(UNION_CPP)
foreach(index RANGE ${last})
    string(JSON relative GET "${UNION_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "union source census file absent: ${absolute}")
    endif()
    list(APPEND UNION_FILES "${absolute}")
    if(absolute MATCHES "\\.(cpp|h)$")
        list(APPEND UNION_CPP "${absolute}")
    endif()
endforeach()
list(LENGTH UNION_FILES before_unique)
list(REMOVE_DUPLICATES UNION_FILES)
list(LENGTH UNION_FILES after_unique)
if(NOT before_unique EQUAL after_unique)
    message(FATAL_ERROR "Duplicate union source census entry")
endif()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT UNION_CPP)
list(SORT ACTUAL_CPP)
if(NOT UNION_CPP STREQUAL ACTUAL_CPP)
    message(FATAL_ERROR "Explicit owned union C++ census incomplete")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../context_decoder/study/source-files.json" DEPENDENCY_CENSUS)
string(JSON dependency_count LENGTH "${DEPENDENCY_CENSUS}")
math(EXPR last_dependency "${dependency_count} - 1")
foreach(index RANGE ${last_dependency})
    string(JSON relative GET "${DEPENDENCY_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../context_decoder/study")
    list(APPEND UNION_FILES "${absolute}")
endforeach()
list(REMOVE_DUPLICATES UNION_FILES)
list(SORT UNION_FILES)
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace context_decision_union {\nstruct UnionCompiledSource { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr UnionCompiledSource union_compiled_sources[] = {\n")
foreach(absolute IN LISTS UNION_FILES)
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "union dependency source absent: ${absolute}")
    endif()
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    string(APPEND WITNESS "{\"${absolute}\",\"${hash}\",${size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
endforeach()
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_hash)
file(SHA256 "${UNION_CLANG_FORMAT}" formatter_hash)
string(JOIN " " flag_string ${UNION_FLAGS})
string(APPEND WITNESS "};\ninline constexpr const char* union_compiler_path = \"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* union_compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* union_compile_flags = \"${flag_string}\";\ninline constexpr const char* union_formatter_path = \"${UNION_CLANG_FORMAT}\";\ninline constexpr const char* union_formatter_sha256 = \"${formatter_hash}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/union_sources.h" "${WITNESS}")
add_library(context_decision_union_source STATIC src/union.cpp tests/controls.cpp)
target_include_directories(context_decision_union_source PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(context_decision_union_source PUBLIC context_inference)
target_compile_options(context_decision_union_source PUBLIC ${UNION_FLAGS})
add_executable(context-decision-union src/main.cpp)
target_link_libraries(context-decision-union PRIVATE context_decision_union_source)
add_custom_target(union-format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${UNION_CLANG_FORMAT}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${UNION_CLANG_FORMAT}" --dry-run --Werror ${UNION_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
add_dependencies(format-check union-format-check)
if(BUILD_TESTING)
    add_test(NAME retained_union_controls COMMAND context-decision-union controls)
    set_tests_properties(retained_union_controls PROPERTIES TIMEOUT 30)
endif()
