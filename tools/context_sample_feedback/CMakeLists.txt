cmake_minimum_required(VERSION 3.25)
project(context_sample_feedback VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SAMPLE_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(SAMPLE_TESTING "${BUILD_TESTING}")
set(BUILD_TESTING OFF)
set(ORIGINAL "/home/jake/Projects/nanopore_basecaller-context-feedback-v2/tools/context_decoder")
set(STUDY "/home/jake/Projects/nanopore_basecaller-context-feedback-v4/tools/context_decoder/study")
add_subdirectory("${STUDY}" "${CMAKE_CURRENT_BINARY_DIR}/study" EXCLUDE_FROM_ALL)
set(BUILD_TESTING "${SAMPLE_TESTING}")
find_program(SAMPLE_FORMATTER NAMES clang-format REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" CENSUS)
string(JSON count LENGTH "${CENSUS}")
math(EXPR last "${count} - 1")
set(SAMPLE_FILES)
set(SAMPLE_CPP)
foreach(index RANGE ${last})
    string(JSON relative GET "${CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    list(APPEND SAMPLE_FILES "${absolute}")
    if(absolute MATCHES "^${CMAKE_CURRENT_SOURCE_DIR}/.*\\.(cpp|h)$")
        list(APPEND SAMPLE_CPP "${absolute}")
    endif()
endforeach()
list(LENGTH SAMPLE_FILES before)
list(REMOVE_DUPLICATES SAMPLE_FILES)
list(LENGTH SAMPLE_FILES after)
if(NOT before EQUAL after)
    message(FATAL_ERROR "Duplicate sample source census entry")
endif()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT SAMPLE_CPP)
list(SORT ACTUAL_CPP)
if(NOT SAMPLE_CPP STREQUAL ACTUAL_CPP)
    message(FATAL_ERROR "Explicit sample C++ source census incomplete")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/engine-files.json" ENGINE_CENSUS)
string(JSON count LENGTH "${ENGINE_CENSUS}")
math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
    string(JSON relative GET "${ENGINE_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${ORIGINAL}")
    list(APPEND SAMPLE_FILES "${absolute}")
endforeach()
file(READ "${STUDY}/source-files.json" DEPENDENCIES)
string(JSON count LENGTH "${DEPENDENCIES}")
math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
    string(JSON relative GET "${DEPENDENCIES}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${STUDY}")
    list(APPEND SAMPLE_FILES "${absolute}")
endforeach()
list(REMOVE_DUPLICATES SAMPLE_FILES)
list(SORT SAMPLE_FILES)
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace context_sample_feedback {\nstruct CompiledSource { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr CompiledSource sample_compiled_sources[] = {\n")
foreach(absolute IN LISTS SAMPLE_FILES)
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "Sample source census missing ${absolute}")
    endif()
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    string(APPEND WITNESS "{\"${absolute}\",\"${hash}\",${size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
endforeach()
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_hash)
file(SHA256 "${SAMPLE_FORMATTER}" formatter_hash)
string(JOIN " " flags ${SAMPLE_FLAGS})
string(APPEND WITNESS "};\ninline constexpr const char* sample_compiler_path = \"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* sample_compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* sample_formatter_path = \"${SAMPLE_FORMATTER}\";\ninline constexpr const char* sample_formatter_sha256 = \"${formatter_hash}\";\ninline constexpr const char* sample_compile_flags = \"${flags}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sample_sources.h" "${WITNESS}")
add_library(old_guarded_engine STATIC "${ORIGINAL}/src/decoder.cpp" "${ORIGINAL}/src/io.cpp" "${ORIGINAL}/src/json.cpp" "${ORIGINAL}/src/guarded.cpp" "${ORIGINAL}/src/guarded_io.cpp")
target_include_directories(old_guarded_engine PUBLIC "${ORIGINAL}/include")
target_link_libraries(old_guarded_engine PUBLIC OpenSSL::Crypto)
target_compile_options(old_guarded_engine PUBLIC ${SAMPLE_FLAGS})
add_library(sample_feedback STATIC src/sample.cpp src/source.cpp src/runtime.cpp tests/controls.cpp)
target_include_directories(sample_feedback PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sample_feedback PUBLIC old_guarded_engine context_inference)
target_compile_options(sample_feedback PUBLIC ${SAMPLE_FLAGS})
add_executable(context-sample-feedback src/main.cpp)
target_link_libraries(context-sample-feedback PRIVATE sample_feedback)
add_executable(context-sample-feedback-observe src/evaluate.cpp)
target_link_libraries(context-sample-feedback-observe PRIVATE sample_feedback context_truth)
add_custom_target(sample-format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${SAMPLE_FORMATTER}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${SAMPLE_FORMATTER}" --dry-run --Werror ${SAMPLE_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_executable(context-sample-feedback-tests tests/main.cpp)
    target_link_libraries(context-sample-feedback-tests PRIVATE sample_feedback)
    add_test(NAME exhaustive_sample_feedback_controls COMMAND context-sample-feedback-tests)
    set_tests_properties(exhaustive_sample_feedback_controls PROPERTIES TIMEOUT 60)
endif()
