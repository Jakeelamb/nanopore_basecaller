cmake_minimum_required(VERSION 3.25)
project(context_model VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(MODEL_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(MODEL_TESTING "${BUILD_TESTING}")
set(BUILD_TESTING OFF)
add_subdirectory("../study" "${CMAKE_CURRENT_BINARY_DIR}/study" EXCLUDE_FROM_ALL)
set(BUILD_TESTING "${MODEL_TESTING}")
find_program(MODEL_CLANG_FORMAT NAMES clang-format REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" MODEL_CENSUS)
string(JSON count LENGTH "${MODEL_CENSUS}")
math(EXPR last "${count} - 1")
set(MODEL_FILES)
set(MODEL_CPP)
foreach(index RANGE ${last})
    string(JSON relative GET "${MODEL_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "Model source census file absent: ${absolute}")
    endif()
    list(APPEND MODEL_FILES "${absolute}")
    if(absolute MATCHES "\\.(cpp|h)$")
        list(APPEND MODEL_CPP "${absolute}")
    endif()
endforeach()
list(LENGTH MODEL_FILES before_unique)
list(REMOVE_DUPLICATES MODEL_FILES)
list(LENGTH MODEL_FILES after_unique)
if(NOT before_unique EQUAL after_unique)
    message(FATAL_ERROR "Duplicate model source census entry")
endif()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/private/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT MODEL_CPP)
list(SORT ACTUAL_CPP)
if(NOT MODEL_CPP STREQUAL ACTUAL_CPP)
    message(FATAL_ERROR "Explicit owned model C++ census incomplete")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../study/source-files.json" DEPENDENCY_CENSUS)
string(JSON dependency_count LENGTH "${DEPENDENCY_CENSUS}")
math(EXPR last_dependency "${dependency_count} - 1")
foreach(index RANGE ${last_dependency})
    string(JSON relative GET "${DEPENDENCY_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../study")
    list(APPEND MODEL_FILES "${absolute}")
endforeach()
list(REMOVE_DUPLICATES MODEL_FILES)
list(SORT MODEL_FILES)
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace context_model {\nstruct ModelCompiledSource { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr ModelCompiledSource model_compiled_sources[] = {\n")
foreach(absolute IN LISTS MODEL_FILES)
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "Model dependency source absent: ${absolute}")
    endif()
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    string(APPEND WITNESS "{\"${absolute}\",\"${hash}\",${size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
endforeach()
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_hash)
file(SHA256 "${MODEL_CLANG_FORMAT}" formatter_hash)
string(JOIN " " flag_string ${MODEL_FLAGS})
string(APPEND WITNESS "};\ninline constexpr const char* model_compiler_path = \"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* model_compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* model_compile_flags = \"${flag_string}\";\ninline constexpr const char* model_formatter_path = \"${MODEL_CLANG_FORMAT}\";\ninline constexpr const char* model_formatter_sha256 = \"${formatter_hash}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/model_sources.h" "${WITNESS}")
add_library(context_model_source STATIC src/model.cpp src/runtime.cpp tests/controls.cpp)
target_include_directories(context_model_source PUBLIC include PRIVATE private "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(context_model_source PUBLIC context_inference)
target_compile_options(context_model_source PUBLIC ${MODEL_FLAGS})
add_executable(context-model src/main.cpp)
target_include_directories(context-model PRIVATE private)
target_link_libraries(context-model PRIVATE context_model_source)
add_custom_target(model-format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${MODEL_CLANG_FORMAT}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${MODEL_CLANG_FORMAT}" --dry-run --Werror ${MODEL_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
add_dependencies(format-check model-format-check)
if(BUILD_TESTING)
    add_executable(context-model-tests tests/main.cpp)
    target_link_libraries(context-model-tests PRIVATE context_model_source)
    add_test(NAME exhaustive_context_model_controls COMMAND context-model-tests)
    set_tests_properties(exhaustive_context_model_controls PROPERTIES TIMEOUT 60)
endif()
