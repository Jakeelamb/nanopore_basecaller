cmake_minimum_required(VERSION 3.25)
project(context_study VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_program(STUDY_CLANG_FORMAT NAMES clang-format REQUIRED)
set(STUDY_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/wire-schema.json" WIRE)
set(WIRE_HEADER "#pragma once\n#include <array>\n#include <cstddef>\n#include <map>\n#include <string>\nnamespace context_study::wire {\n")
foreach(group IN ITEMS candidate truth)
    string(JSON record_count LENGTH "${WIRE}" "${group}_records")
    math(EXPR last_record "${record_count} - 1")
    set(widths "")
    foreach(record_index RANGE ${last_record})
        string(JSON record MEMBER "${WIRE}" "${group}_records" ${record_index})
        string(JSON field_count LENGTH "${WIRE}" "${group}_records" "${record}")
        math(EXPR width "${field_count} + 1")
        string(APPEND widths "{\"${record}\",${width}},")
        string(APPEND WIRE_HEADER "namespace ${record} {\n")
        math(EXPR last_field "${field_count} - 1")
        foreach(field_index RANGE ${last_field})
            string(JSON field GET "${WIRE}" "${group}_records" "${record}" ${field_index})
            math(EXPR position "${field_index} + 1")
            string(APPEND WIRE_HEADER "inline constexpr std::size_t ${field} = ${position};\n")
        endforeach()
        string(APPEND WIRE_HEADER "}\n")
    endforeach()
    string(APPEND WIRE_HEADER "inline const std::map<std::string,std::size_t> ${group}_widths = {${widths}};\n")
endforeach()
string(JSON geometry_count LENGTH "${WIRE}" geometry_recipe case_fields)
math(EXPR last_geometry "${geometry_count} - 1")
string(APPEND WIRE_HEADER "inline constexpr std::array<std::size_t,${geometry_count}> geometry_indices = {\n")
foreach(index RANGE ${last_geometry})
    string(JSON field GET "${WIRE}" geometry_recipe case_fields ${index})
    string(APPEND WIRE_HEADER "CASE::${field},\n")
endforeach()
string(APPEND WIRE_HEADER "};\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wire_fields.h" "${WIRE_HEADER}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" CENSUS)
string(JSON source_count LENGTH "${CENSUS}")
math(EXPR last_source "${source_count} - 1")
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace context_study {\nstruct CompiledSource { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr CompiledSource compiled_sources[] = {\n")
set(OWNED_CPP)
foreach(index RANGE ${last_source})
    string(JSON relative GET "${CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "Source census file is absent: ${absolute}")
    endif()
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    string(APPEND WITNESS "{\"${absolute}\",\"${hash}\",${size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
    if(absolute MATCHES "^${CMAKE_CURRENT_SOURCE_DIR}/.*\\.(cpp|h)$")
        list(APPEND OWNED_CPP "${absolute}")
    endif()
endforeach()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/private/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT OWNED_CPP)
list(SORT ACTUAL_CPP)
if(NOT OWNED_CPP STREQUAL ACTUAL_CPP)
    message(FATAL_ERROR "Explicit owned C++ census is incomplete or duplicated")
endif()
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_hash)
file(SHA256 "${STUDY_CLANG_FORMAT}" formatter_hash)
string(JOIN " " flag_string ${STUDY_FLAGS})
string(APPEND WITNESS "};\ninline constexpr const char* compiler_path = \"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* compile_flags = \"${flag_string}\";\ninline constexpr const char* formatter_path = \"${STUDY_CLANG_FORMAT}\";\ninline constexpr const char* formatter_sha256 = \"${formatter_hash}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${WITNESS}")
add_library(context_inference STATIC src/historical.cpp src/packet.cpp src/replay_output.cpp)
target_include_directories(context_inference PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(context_inference PUBLIC OpenSSL::Crypto)
target_compile_options(context_inference PUBLIC ${STUDY_FLAGS})
add_library(context_truth STATIC src/oracle.cpp src/atlas.cpp tests/controls.cpp)
target_include_directories(context_truth PUBLIC private PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(context_truth PUBLIC context_inference)
add_executable(context-infer src/infer_main.cpp)
target_link_libraries(context-infer PRIVATE context_inference)
add_executable(context-atlas src/main.cpp)
target_link_libraries(context-atlas PRIVATE context_truth)
add_custom_target(format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${STUDY_CLANG_FORMAT}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${STUDY_CLANG_FORMAT}" --dry-run --Werror ${OWNED_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_executable(context-study-tests tests/main.cpp)
    target_link_libraries(context-study-tests PRIVATE context_truth)
    add_test(NAME exhaustive_context_study_controls COMMAND context-study-tests)
    set_tests_properties(exhaustive_context_study_controls PROPERTIES TIMEOUT 30)
endif()
