cmake_minimum_required(VERSION 3.25)
set(PREFIX_COMPILER "/home/jake/Projects/nanopore_basecaller/software/source-build-qualification-20261023/toolchain-gcc15/usr/bin/g++-15")
set(CMAKE_CXX_COMPILER "${PREFIX_COMPILER}" CACHE FILEPATH "Retained native compiler")
project(context_prefix VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
if(NOT CMAKE_CXX_COMPILER STREQUAL PREFIX_COMPILER)
    message(FATAL_ERROR "Prefix helper requires the retained GCC15 compiler")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_COMPILER_LAUNCHER "" CACHE STRING "Disable shared compiler caches" FORCE)
set(PREFIX_UV "/home/jake/.local/bin/uv" CACHE FILEPATH "Qualified UV orchestration")
set(PREFIX_ENV "/home/jake/Projects/nanopore_basecaller-retrieval/.venv" CACHE PATH "Existing project environment")
execute_process(COMMAND "${CMAKE_COMMAND}" -E env "UV_PROJECT_ENVIRONMENT=${PREFIX_ENV}" "${PREFIX_UV}" run --no-sync python "${CMAKE_CURRENT_SOURCE_DIR}/derive_build.py" --source "${CMAKE_CURRENT_SOURCE_DIR}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.."
    RESULT_VARIABLE derived COMMAND_ERROR_IS_FATAL ANY)
include("${CMAKE_CURRENT_BINARY_DIR}/generated/native.cmake")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PREFIX_CLOSURE_FILES})
add_library(context_prefix_native STATIC src/prefix.cpp src/native.cpp tests/controls.cpp)
target_include_directories(context_prefix_native PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(context_prefix_native SYSTEM PRIVATE ${PREFIX_NATIVE_SYSTEM_INCLUDES})
target_compile_options(context_prefix_native PUBLIC ${PREFIX_NATIVE_OPTIONS})
target_link_libraries(context_prefix_native PUBLIC ${PREFIX_NATIVE_LIBRARIES})
add_executable(context-prefix src/main.cpp)
target_include_directories(context-prefix PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(context-prefix PRIVATE context_prefix_native)
target_link_options(context-prefix PRIVATE ${PREFIX_NATIVE_LINK_OPTIONS})
add_custom_target(format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${PREFIX_FORMATTER}" "-DEXPECTED=${PREFIX_FORMATTER_SHA256}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${PREFIX_FORMATTER}" --dry-run --Werror ${PREFIX_OWNED_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_executable(context-prefix-tests tests/main.cpp)
    target_link_libraries(context-prefix-tests PRIVATE context_prefix_native)
    add_test(NAME prefix_cpu_contract_controls COMMAND context-prefix-tests)
    set_tests_properties(prefix_cpu_contract_controls PROPERTIES TIMEOUT 60)
endif()
