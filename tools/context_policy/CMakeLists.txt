cmake_minimum_required(VERSION 3.25)
project(context_policy VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FEEDBACK_BASE_ROOT "" CACHE PATH "Pinned read-only PR15 source root")
set(FEEDBACK_CLANG_FORMAT "/usr/bin/clang-format" CACHE FILEPATH "Pinned formatter")
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
file(READ "${FEEDBACK_BASE_ROOT}/tools/uncertainty_feedback/dependencies.json" dependencies)
string(JSON json_header GET "${dependencies}" nlohmann_json path)
string(JSON json_expected GET "${dependencies}" nlohmann_json sha256)
file(SHA256 "${json_header}" json_actual)
if(NOT json_actual STREQUAL json_expected)
    message(FATAL_ERROR "Pinned nlohmann JSON header changed")
endif()
file(SHA256 "${FEEDBACK_CLANG_FORMAT}" format_actual)
if(NOT format_actual STREQUAL "cd50c00c4b3475329257d717f9e778769d00d1eff5c03436ebc39b023ddd6394")
    message(FATAL_ERROR "Pinned clang-format changed")
endif()
get_filename_component(json_include "${json_header}" DIRECTORY)
get_filename_component(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
set(owned_cpp include/context_policy/policy.h include/context_policy/protocol.h src/policy.cpp
    src/protocol.cpp src/main.cpp tests/fixtures.h tests/controls.cpp)
file(GLOB_RECURSE actual_cpp RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS
    include/*.h src/*.cpp tests/*.h tests/*.cpp)
set(expected_cpp ${owned_cpp})
list(SORT actual_cpp)
list(SORT expected_cpp)
if(NOT actual_cpp STREQUAL expected_cpp)
    message(FATAL_ERROR "Owned C++ source census incomplete")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" source_files)
string(JSON source_count LENGTH "${source_files}")
math(EXPR source_last "${source_count}-1")
set(source_closure "${json_header}")
foreach(index RANGE 0 ${source_last})
    string(JSON relative GET "${source_files}" ${index})
    list(APPEND source_closure "${source_root}/${relative}")
endforeach()
set(witness "#pragma once\n#include <cstdint>\nnamespace context_policy {\nstruct Source {const char* path; std::uintmax_t size; const char* sha256;};\ninline constexpr Source compiled_sources[] = {\n")
foreach(path IN LISTS source_closure)
    file(SHA256 "${path}" hash)
    file(SIZE "${path}" size)
    string(APPEND witness "{\"${path}\",${size},\"${hash}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
endforeach()
string(APPEND witness "};\ninline constexpr const char* source_root_path=\"${source_root}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${witness}")
add_library(context-policy-core src/policy.cpp)
target_include_directories(context-policy-core PUBLIC include)
add_library(context-policy-protocol src/protocol.cpp)
target_link_libraries(context-policy-protocol PUBLIC context-policy-core OpenSSL::Crypto)
target_include_directories(context-policy-protocol PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(context-policy-protocol SYSTEM PUBLIC "${json_include}")
add_executable(context-policy src/main.cpp)
target_link_libraries(context-policy PRIVATE context-policy-protocol)
add_executable(context-policy-controls tests/controls.cpp)
target_link_libraries(context-policy-controls PRIVATE context-policy-protocol)
foreach(target context-policy-core context-policy-protocol context-policy context-policy-controls)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion
        -Wshadow -Werror -fno-fast-math -ffp-contract=off)
endforeach()
set(format_sources)
foreach(path IN LISTS owned_cpp)
    list(APPEND format_sources "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
add_custom_target(format-check COMMAND "${FEEDBACK_CLANG_FORMAT}" --dry-run --Werror ${format_sources}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    foreach(name separation runner_up fixed monotonic novel wrong_inside_budget ties empty_context
            numerical_rejections map_oracle protocol source_guard)
        add_test(NAME policy_${name} COMMAND context-policy-controls ${name})
        set_tests_properties(policy_${name} PROPERTIES TIMEOUT 30)
    endforeach()
endif()
