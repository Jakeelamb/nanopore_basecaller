cmake_minimum_required(VERSION 3.25)
project(indel_feedback VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FEEDBACK_BASE_ROOT "/home/jake/Projects/nanopore_basecaller-context-feedback-targeting-results" CACHE PATH "Read-only PR15 source root")
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_program(FEEDBACK_CLANG_FORMAT NAMES clang-format REQUIRED)
file(SHA256 "${FEEDBACK_CLANG_FORMAT}" format_actual)
if(NOT format_actual STREQUAL "cd50c00c4b3475329257d717f9e778769d00d1eff5c03436ebc39b023ddd6394")
    message(FATAL_ERROR "Pinned clang-format changed")
endif()
set(json_header "/home/jake/Projects/tools/clang-uml-src/thirdparty/nlohmann/json.hpp")
file(SHA256 "${json_header}" json_actual)
if(NOT json_actual STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
    message(FATAL_ERROR "Pinned nlohmann JSON header changed")
endif()
get_filename_component(json_include "${json_header}" DIRECTORY)
set(owned_cpp include/indel_feedback/core.h src/core.cpp src/classifier.cpp src/proposals.cpp src/io.cpp src/main.cpp src/controls.cpp)
file(GLOB_RECURSE actual_cpp RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS include/*.h src/*.cpp)
set(expected_cpp ${owned_cpp})
list(SORT actual_cpp)
list(SORT expected_cpp)
if(NOT actual_cpp STREQUAL expected_cpp)
    message(FATAL_ERROR "Owned source census differs")
endif()
set(witness "#pragma once\n#include <cstdint>\nnamespace indel_feedback {\nstruct Source { const char* path; std::uintmax_t size; const char* sha256; };\ninline constexpr Source compiled_sources[] = {\n")
set(source_closure ${owned_cpp} CMakeLists.txt qualify.py
    ../../scripts/indel-feedback.py ../../tests/test_indel_feedback.py
    ../../config/indel-feedback-dev-v1.json)
foreach(relative IN LISTS source_closure)
    get_filename_component(path "${CMAKE_CURRENT_SOURCE_DIR}/${relative}" ABSOLUTE)
    file(SHA256 "${path}" hash)
    file(SIZE "${path}" size)
    string(APPEND witness "{\"${path}\",${size},\"${hash}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
endforeach()
get_filename_component(source_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
string(APPEND witness "{\"${json_header}\",919975,\"${json_actual}\"},\n};\ninline constexpr const char* source_root_path=\"${source_root}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source_witness.h" "${witness}")
add_library(indel-feedback-core STATIC src/core.cpp src/classifier.cpp src/proposals.cpp src/io.cpp)
target_include_directories(indel-feedback-core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(indel-feedback-core SYSTEM PUBLIC "${json_include}")
target_link_libraries(indel-feedback-core PUBLIC OpenSSL::Crypto)
add_executable(indel-feedback src/main.cpp src/controls.cpp)
target_link_libraries(indel-feedback PRIVATE indel-feedback-core)
foreach(target indel-feedback-core indel-feedback)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
endforeach()
set(format_sources)
foreach(path IN LISTS owned_cpp)
    list(APPEND format_sources "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
add_custom_target(format-check COMMAND "${FEEDBACK_CLANG_FORMAT}" --dry-run --Werror ${format_sources} COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_test(NAME indel_controls COMMAND indel-feedback controls)
    set_tests_properties(indel_controls PROPERTIES TIMEOUT 60 WORKING_DIRECTORY "${source_root}")
endif()
