cmake_minimum_required(VERSION 3.25)
project(context_decision VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(DECISION_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(DECISION_TESTING "${BUILD_TESTING}")
set(BUILD_TESTING OFF)
add_subdirectory("../context_decoder/study" "${CMAKE_CURRENT_BINARY_DIR}/study" EXCLUDE_FROM_ALL)
set(BUILD_TESTING "${DECISION_TESTING}")
find_program(DECISION_CLANG_FORMAT NAMES clang-format REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" DECISION_CENSUS)
string(JSON count LENGTH "${DECISION_CENSUS}")
math(EXPR last "${count} - 1")
set(DECISION_FILES)
set(DECISION_CPP)
foreach(index RANGE ${last})
    string(JSON relative GET "${DECISION_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "decision source census file absent: ${absolute}")
    endif()
    list(APPEND DECISION_FILES "${absolute}")
    if(absolute MATCHES "\\.(cpp|h)$")
        list(APPEND DECISION_CPP "${absolute}")
    endif()
endforeach()
list(LENGTH DECISION_FILES before_unique)
list(REMOVE_DUPLICATES DECISION_FILES)
list(LENGTH DECISION_FILES after_unique)
if(NOT before_unique EQUAL after_unique)
    message(FATAL_ERROR "Duplicate decision source census entry")
endif()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/private/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT DECISION_CPP)
list(SORT ACTUAL_CPP)
if(NOT DECISION_CPP STREQUAL ACTUAL_CPP)
    message(FATAL_ERROR "Explicit owned decision C++ census incomplete")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../context_decoder/study/source-files.json" DEPENDENCY_CENSUS)
string(JSON dependency_count LENGTH "${DEPENDENCY_CENSUS}")
math(EXPR last_dependency "${dependency_count} - 1")
foreach(index RANGE ${last_dependency})
    string(JSON relative GET "${DEPENDENCY_CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../context_decoder/study")
    list(APPEND DECISION_FILES "${absolute}")
endforeach()
list(REMOVE_DUPLICATES DECISION_FILES)
list(SORT DECISION_FILES)
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace context_decision {\nstruct DecisionCompiledSource { const char* path; const char* sha256; std::uintmax_t size; };\ninline constexpr DecisionCompiledSource decision_compiled_sources[] = {\n")
foreach(absolute IN LISTS DECISION_FILES)
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "decision dependency source absent: ${absolute}")
    endif()
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    string(APPEND WITNESS "{\"${absolute}\",\"${hash}\",${size}},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
endforeach()
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_hash)
file(SHA256 "${DECISION_CLANG_FORMAT}" formatter_hash)
string(JOIN " " flag_string ${DECISION_FLAGS})
string(APPEND WITNESS "};\ninline constexpr const char* decision_compiler_path = \"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* decision_compiler_sha256 = \"${compiler_hash}\";\ninline constexpr const char* decision_compile_flags = \"${flag_string}\";\ninline constexpr const char* decision_formatter_path = \"${DECISION_CLANG_FORMAT}\";\ninline constexpr const char* decision_formatter_sha256 = \"${formatter_hash}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/decision_sources.h" "${WITNESS}")
add_library(context_decision_source STATIC src/decision.cpp src/runtime.cpp tests/controls.cpp)
target_include_directories(context_decision_source PUBLIC include PRIVATE private "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(context_decision_source PUBLIC context_inference)
target_compile_options(context_decision_source PUBLIC ${DECISION_FLAGS})
add_executable(context-decision src/main.cpp)
target_include_directories(context-decision PRIVATE private)
target_link_libraries(context-decision PRIVATE context_decision_source)
add_custom_target(decision-format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${DECISION_CLANG_FORMAT}" "-DEXPECTED=${formatter_hash}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${DECISION_CLANG_FORMAT}" --dry-run --Werror ${DECISION_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
add_dependencies(format-check decision-format-check)
if(BUILD_TESTING)
    add_executable(context-decision-tests tests/main.cpp)
    target_link_libraries(context-decision-tests PRIVATE context_decision_source)
    add_test(NAME exhaustive_context_decision_controls COMMAND context-decision-tests)
    set_tests_properties(exhaustive_context_decision_controls PROPERTIES TIMEOUT 60)
endif()
