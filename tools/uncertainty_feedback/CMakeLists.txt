cmake_minimum_required(VERSION 3.25)
project(uncertainty_feedback VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_program(TARGETING_CLANG_FORMAT NAMES clang-format REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.json" dependencies)
string(JSON json_header GET "${dependencies}" nlohmann_json path)
string(JSON json_expected GET "${dependencies}" nlohmann_json sha256)
file(SHA256 "${json_header}" json_actual)
if(NOT json_actual STREQUAL json_expected)
    message(FATAL_ERROR "Pinned nlohmann JSON header changed")
endif()
get_filename_component(json_include "${json_header}" DIRECTORY)
set(owned_cpp include/uncertainty_feedback/core.h src/core.cpp src/plan.cpp src/graph.cpp src/frames.cpp
    src/emission.cpp src/exposure.cpp src/main.cpp src/controls.cpp)
file(GLOB_RECURSE actual_cpp RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS include/*.h src/*.cpp)
set(expected_cpp ${owned_cpp})
list(SORT actual_cpp)
list(SORT expected_cpp)
if(NOT actual_cpp STREQUAL expected_cpp)
    message(FATAL_ERROR "Owned C++ source census incomplete")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" source_json)
string(JSON source_count LENGTH "${source_json}")
math(EXPR source_last "${source_count}-1")
set(source_closure "${json_header}")
foreach(index RANGE 0 ${source_last})
    string(JSON relative GET "${source_json}" ${index})
    get_filename_component(absolute "${CMAKE_CURRENT_SOURCE_DIR}/${relative}" ABSOLUTE)
    list(APPEND source_closure "${absolute}")
endforeach()
set(witness "#pragma once\n#include <cstdint>\nnamespace uncertainty_feedback {\nstruct Source {const char* path; std::uintmax_t size; const char* sha256;};\ninline constexpr Source compiled_sources[] = {\n")
foreach(path IN LISTS source_closure)
    file(SHA256 "${path}" hash)
    file(SIZE "${path}" size)
    string(APPEND witness "{\"${path}\",${size},\"${hash}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
endforeach()
file(REAL_PATH "${CMAKE_CXX_COMPILER}" compiler)
file(SHA256 "${compiler}" compiler_hash)
file(SIZE "${compiler}" compiler_size)
string(JSON expected_compiler GET "${dependencies}" cxx_compiler path)
string(JSON expected_compiler_hash GET "${dependencies}" cxx_compiler sha256)
if(NOT compiler STREQUAL expected_compiler OR NOT compiler_hash STREQUAL expected_compiler_hash)
    message(FATAL_ERROR "Exact declared C++ compiler changed")
endif()
get_filename_component(connector_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
string(APPEND witness "};\ninline constexpr const char* source_root_path=\"${connector_root}\";\ninline constexpr const char* compiler_path=\"${compiler}\";\ninline constexpr const char* compiler_sha256=\"${compiler_hash}\";\ninline constexpr std::uintmax_t compiler_size=${compiler_size};\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${witness}")
add_executable(uncertainty-feedback src/core.cpp src/plan.cpp src/graph.cpp src/frames.cpp src/emission.cpp src/exposure.cpp src/main.cpp src/controls.cpp)
target_include_directories(uncertainty-feedback PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(uncertainty-feedback SYSTEM PRIVATE "${json_include}")
target_link_libraries(uncertainty-feedback PRIVATE OpenSSL::Crypto)
target_compile_options(uncertainty-feedback PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
set(format_sources)
foreach(path IN LISTS owned_cpp)
    list(APPEND format_sources "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
add_custom_target(format-check COMMAND "${TARGETING_CLANG_FORMAT}" --dry-run --Werror ${format_sources} COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_test(NAME targeting_synthetic_controls COMMAND uncertainty-feedback controls)
    set_tests_properties(targeting_synthetic_controls PROPERTIES TIMEOUT 60)
endif()
