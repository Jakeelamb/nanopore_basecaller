cmake_minimum_required(VERSION 3.25)
project(context_uncertainty VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_program(UNCERTAINTY_FORMATTER NAMES clang-format REQUIRED)
set(UNCERTAINTY_FLAGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror -fno-fast-math -ffp-contract=off)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-files.json" CENSUS)
string(JSON count LENGTH "${CENSUS}")
math(EXPR last "${count} - 1")
set(ALL_FILES)
set(OWNED_CPP)
foreach(index RANGE ${last})
    string(JSON relative GET "${CENSUS}" ${index})
    get_filename_component(absolute "${relative}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(NOT EXISTS "${absolute}")
        message(FATAL_ERROR "Source census file absent: ${absolute}")
    endif()
    list(APPEND ALL_FILES "${absolute}")
    if(absolute MATCHES "\\.(cpp|h)$")
        list(APPEND OWNED_CPP "${absolute}")
    endif()
endforeach()
list(LENGTH ALL_FILES before)
list(REMOVE_DUPLICATES ALL_FILES)
list(LENGTH ALL_FILES after)
if(NOT before EQUAL after)
    message(FATAL_ERROR "Duplicate source census entry")
endif()
file(GLOB_RECURSE ACTUAL_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/tests/*.cpp")
list(SORT ACTUAL_CPP)
list(SORT OWNED_CPP)
if(NOT ACTUAL_CPP STREQUAL OWNED_CPP)
    message(FATAL_ERROR "Explicit C++ source census incomplete")
endif()
list(SORT ALL_FILES)
set(WITNESS "#pragma once\n#include <cstdint>\nnamespace context_uncertainty {\nstruct CompiledSource { const char* path; std::uintmax_t size; const char* sha256; };\ninline constexpr CompiledSource compiled_sources[] = {\n")
foreach(absolute IN LISTS ALL_FILES)
    file(SHA256 "${absolute}" hash)
    file(SIZE "${absolute}" size)
    string(APPEND WITNESS "{\"${absolute}\",${size},\"${hash}\"},\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${absolute}")
endforeach()
file(SHA256 "${CMAKE_CXX_COMPILER}" COMPILER_SHA)
file(SHA256 "${UNCERTAINTY_FORMATTER}" FORMATTER_SHA)
string(JOIN " " FLAGS_STRING ${UNCERTAINTY_FLAGS})
string(APPEND WITNESS "};\ninline constexpr const char* compiler_path=\"${CMAKE_CXX_COMPILER}\";\ninline constexpr const char* compiler_sha256=\"${COMPILER_SHA}\";\ninline constexpr const char* compile_flags=\"${FLAGS_STRING}\";\ninline constexpr const char* formatter_path=\"${UNCERTAINTY_FORMATTER}\";\ninline constexpr const char* formatter_sha256=\"${FORMATTER_SHA}\";\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_sources.h" "${WITNESS}")
add_library(context_uncertainty_source STATIC src/posterior.cpp src/runtime.cpp tests/controls.cpp)
target_include_directories(context_uncertainty_source PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(context_uncertainty_source PUBLIC ${UNCERTAINTY_FLAGS})
target_link_libraries(context_uncertainty_source PUBLIC OpenSSL::Crypto)
add_executable(context-uncertainty src/main.cpp)
target_link_libraries(context-uncertainty PRIVATE context_uncertainty_source)
add_custom_target(format-check
    COMMAND "${CMAKE_COMMAND}" "-DFORMATTER=${UNCERTAINTY_FORMATTER}" "-DEXPECTED=${FORMATTER_SHA}" -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_formatter.cmake"
    COMMAND "${UNCERTAINTY_FORMATTER}" --dry-run --Werror ${OWNED_CPP}
    COMMAND_EXPAND_LISTS VERBATIM)
if(BUILD_TESTING)
    add_executable(context-uncertainty-tests tests/main.cpp)
    target_link_libraries(context-uncertainty-tests PRIVATE context_uncertainty_source)
    add_test(NAME exhaustive_uncertainty_controls COMMAND context-uncertainty-tests)
    set_tests_properties(exhaustive_uncertainty_controls PROPERTIES TIMEOUT 60)
endif()
